\section*{Scenario-Gated DBM}

\textbf{Current DBM.}

\[
r_{ijC}=\langle u_i\odot\phi_C,v_j\rangle,\qquad
\left[r_{ijC},\;\frac{r_{ijC}+r_{ikC}}{2}+\ell_i,\;r_{ikC}\right],
\qquad
\mathcal L_{\mathrm{base}}=\operatorname{CE}(\text{logits},\text{trit}).
\]

\textbf{Scenario gate.} A frozen text embedding $z_s$ produces a positive per-axis gate and tie head:

\[
g_s=\exp(Wz_s),\qquad b_s=w_b^\top z_s,\qquad
r_{ijCs}=\langle u_i\odot\phi_C\odot g_s,v_j\rangle.
\]

\[
\text{logits}_{ijkCs}
=\left[r_{ijCs},\;\frac{r_{ijCs}+r_{ikCs}}{2}+\ell_i+b_s,\;r_{ikCs}\right],
\qquad
\mathcal L_{\mathrm{trit}}=\operatorname{CE}(\text{logits},\text{trit}).
\]

Centred logits are $[(r_j-r_k)/2,\;\ell_i+b_s,\;-(r_j-r_k)/2]$: $g_s$ selects axes; $b_s$ separates the prompt's tie rate from the judge's $\ell_i$. Positive $g_s$ scales axes without changing their signs.

\section*{Other training targets}

\textbf{Numerical ratings.} Predict centred ratings, scaled per judge:

\[
\widetilde y_{ijCs}
=\frac{y_{ijCs}-\frac{1}{M}\sum_k y_{ikCs}}{\kappa_i},
\qquad
\widehat y_{ijCs}=\langle u_i\odot\phi_C\odot g_s,v_j-\bar v\rangle,
\qquad
\mathcal L_{\mathrm{rating}}=(\widehat y_{ijCs}-\widetilde y_{ijCs})^2.
\]

Fit scenario-level ratings; averaging over $s$ removes the signal for $g_s$.
